cmake_minimum_required(VERSION 3.16)
project(dp32_fp16_model LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dp32 src/dp32.cpp)
target_include_directories(dp32 PUBLIC include)
add_executable(dp32_demo src/demo.cpp)
target_link_libraries(dp32_demo PRIVATE dp32)
add_executable(dp32_testbench tests/testbench.cpp)
target_link_libraries(dp32_testbench PRIVATE dp32)
enable_testing()
add_test(NAME dp32_all COMMAND dp32_testbench "${CMAKE_CURRENT_SOURCE_DIR}/tests/golden_vectors.txt")
